# PostgreSQL Schema (required tables)
```sql
-- Audit log table
CREATE TABLE audit_events (
    id UUID PRIMARY KEY,
    event_type VARCHAR(50) NOT NULL,
    user_id TEXT REFERENCES users(username),
    ip_address INET,
    user_agent TEXT,
    resource_type VARCHAR(30) NOT NULL,
    resource_id TEXT NOT NULL,
    action VARCHAR(50) NOT NULL,
    details JSONB,
    timestamp TIMESTAMPTZ DEFAULT NOW()
);

-- GIN index for JSONB queries
CREATE INDEX idx_audit_details ON audit_events USING GIN (details);
CREATE INDEX idx_audit_timestamp ON audit_events (timestamp DESC);

-- View for common query patterns
CREATE VIEW rule_audit_history AS
SELECT 
    ae.id, ae.event_type, ae.user_id, ae.timestamp,
    ae.action, ae.details,
    r.name as rule_name, r.version
FROM audit_events ae
JOIN rules r ON ae.resource_id = r.id::text
WHERE ae.resource_type = 'rule'
ORDER BY ae.timestamp DESC;
```
